packaging_defects.command_line

Command-line entry point: packaging-defects <command>.

Command What it does
train Train the hand-written detector and score it on validation.
evaluate Score the trained detector once on the test photographs.
baseline Train and score the Ultralytics alternative the same way.
compare Put the two sets of test results side by side.
study Measure how each design choice changes the validation score.
profile Time the detector and find where the time goes.

The roles of the three parts of the dataset follow the unit notes: training photographs fit the network, validation photographs choose settings and the confidence threshold, and test photographs are used once, at the end, for the reported result.

 36def score(
 37    detections: list[tuple],
 38    split: Split,
 39    class_names: list[str],
 40    confidence_threshold: float | None = None,
 41) -> dict:
 42    """Score detections on one part of the dataset.
 43
 44    Parameters
 45    ----------
 46    detections : list of (boxes, scores, classes)
 47        One entry per photograph of ``split``.
 48    split : Split
 49        The photographs and their labels.
 50    class_names : list of str
 51    confidence_threshold : float, optional
 52        Threshold chosen beforehand on validation data.  ``None`` chooses the
 53        best threshold on this data, which is only appropriate for validation.
 54
 55    Returns
 56    -------
 57    dict
 58        Detection scores at the usual overlap of 0.5, average precision at
 59        the stricter overlap of 0.75, and the defective-or-clean verdicts.
 60    """
 61    detection = evaluate_detections(
 62        detections, split.boxes, split.classes, class_names, 0.5, confidence_threshold
 63    )
 64    strict = evaluate_detections(detections, split.boxes, split.classes, class_names, 0.75)
 65    return {
 66        "photographs": len(split),
 67        "detection": detection,
 68        "average_precision_at_overlap_0.75": strict["any_defect"]["average_precision"],
 69        "verdicts": photograph_verdicts(
 70            detections, split.boxes, detection["any_defect"]["threshold"]
 71        ),
 72    }

 75def summary_line(label: str, result: dict) -> str:
 76    """Format the headline numbers of a :func:`score` result on one line."""
 77    any_defect = result["detection"]["any_defect"]
 78    return (
 79        f"{label}: average precision {any_defect['average_precision']:.3f}, "
 80        f"precision {any_defect['precision']:.2f}, recall {any_defect['recall']:.2f} "
 81        f"at confidence {any_defect['threshold']:.2f}; "
 82        f"balanced accuracy of verdicts {result['verdicts']['balanced_accuracy']:.2f}"
 83    )

 95def command_train(arguments: argparse.Namespace) -> None:
 96    """Train the hand-written detector and score it on validation."""
 97    class_names = read_class_names(arguments.dataset)
 98    data = load_dataset(arguments.dataset)
 99    detector = GridDetector(class_names, Settings(seed=arguments.seed))
100
101    start = time.perf_counter()
102    history = detector.fit(data["train"], arguments.cache / "train.npz", report=print)
103    training_seconds = time.perf_counter() - start
104    detector.save(arguments.artefacts / DETECTOR_FILE)
105    plots.plot_training_history(history, arguments.figures / "training-history.svg")
106
107    detections = detector.detect(data["valid"].image_paths, arguments.cache / "valid.npz")
108    result = score(detections, data["valid"], class_names)
109    result.update(
110        method="hand-written grid regression",
111        split="valid",
112        training_seconds=training_seconds,
113        parameters=detector.network.number_of_parameters(),
114        passes=len(history["loss"]),
115    )
116    write_json(arguments.artefacts / "detector-validation.json", result)
117    print(summary_line("validation", result))

120def command_evaluate(arguments: argparse.Namespace) -> None:
121    """Score the trained detector once on the test photographs."""
122    class_names = read_class_names(arguments.dataset)
123    test = load_dataset(arguments.dataset)["test"]
124    detector = GridDetector.load(arguments.artefacts / DETECTOR_FILE)
125    validation = json.loads((arguments.artefacts / "detector-validation.json").read_text())
126    threshold = validation["detection"]["any_defect"]["threshold"]
127
128    # No saved features here, so the timing includes reading the photographs.
129    start = time.perf_counter()
130    detections = detector.detect(test.image_paths)
131    seconds = (time.perf_counter() - start) / len(test)
132
133    result = score(detections, test, class_names, threshold)
134    result.update(
135        method="hand-written grid regression",
136        split="test",
137        seconds_per_photograph=seconds,
138        parameters=detector.network.number_of_parameters(),
139        training_seconds=validation["training_seconds"],
140    )
141    write_json(arguments.artefacts / "detector-test.json", result)
142    plots.plot_precision_recall(
143        {"hand-written detector": result["detection"]["any_defect"]},
144        arguments.figures / "precision-recall.svg",
145    )
146    # Show evenly spaced photographs, not hand-picked successes.
147    shown = list(range(0, len(test), max(1, len(test) // 6)))[:6]
148    plots.plot_examples(
149        [test.image_paths[i] for i in shown],
150        [detections[i] for i in shown],
151        [test.boxes[i] for i in shown],
152        threshold,
153        arguments.figures / "example-detections.png",
154    )
155    print(summary_line("test", result))

158def command_baseline(arguments: argparse.Namespace) -> None:
159    """Train and score the Ultralytics alternative the same way."""
160    # Imported here so the other commands do not have to load PyTorch.
161    from ultralytics import YOLO
162
163    from . import baseline
164
165    class_names = read_class_names(arguments.dataset)
166    data = load_dataset(arguments.dataset)
167    if arguments.weights:
168        model = YOLO(arguments.weights)
169        # Ultralytics logs the running total of training time in a table
170        # beside the weights folder; use it if it is there.
171        log = Path(arguments.weights).resolve().parent.parent / "results.csv"
172        training_seconds = None
173        if log.is_file():
174            header, *rows = log.read_text().strip().splitlines()
175            training_seconds = float(rows[-1].split(",")[header.split(",").index("time")])
176    else:
177        start = time.perf_counter()
178        model = baseline.train(
179            arguments.dataset / "data.yaml",
180            arguments.epochs,
181            device=arguments.device,
182            seed=arguments.seed,
183            project=str(Path("runs").resolve()),
184            name="baseline",
185            exist_ok=True,
186        )
187        training_seconds = time.perf_counter() - start
188
189    validation = score(
190        baseline.detect(model, data["valid"].image_paths), data["valid"], class_names
191    )
192    threshold = validation["detection"]["any_defect"]["threshold"]
193    start = time.perf_counter()
194    detections = baseline.detect(model, data["test"].image_paths)
195    seconds = (time.perf_counter() - start) / len(data["test"])
196
197    result = score(detections, data["test"], class_names, threshold)
198    for content, split in ((validation, "valid"), (result, "test")):
199        content.update(
200            method="Ultralytics YOLOv8 nano",
201            split=split,
202            parameters=int(sum(p.numel() for p in model.model.parameters())),
203            training_seconds=training_seconds,
204        )
205    result["seconds_per_photograph"] = seconds
206    write_json(arguments.artefacts / "baseline-validation.json", validation)
207    write_json(arguments.artefacts / "baseline-test.json", result)
208    print(summary_line("validation", validation))
209    print(summary_line("test", result))

212def command_compare(arguments: argparse.Namespace) -> None:
213    """Put the two sets of test results side by side."""
214    results = {
215        "Hand-written detector": json.loads(
216            (arguments.artefacts / "detector-test.json").read_text()
217        ),
218        "Ultralytics YOLOv8 nano": json.loads(
219            (arguments.artefacts / "baseline-test.json").read_text()
220        ),
221    }
222
223    def cell(value, pattern: str) -> str:
224        return "not recorded" if value is None else pattern.format(value)
225
226    rows = [
227        ("Average precision, any defect (overlap 0.5)",
228         lambda r: cell(r["detection"]["any_defect"]["average_precision"], "{:.3f}")),
229        ("Average precision, any defect (overlap 0.75)",
230         lambda r: cell(r["average_precision_at_overlap_0.75"], "{:.3f}")),
231        ("Mean average precision over classes (overlap 0.5)",
232         lambda r: cell(r["detection"]["mean_average_precision"], "{:.3f}")),
233        ("Precision at chosen confidence",
234         lambda r: cell(r["detection"]["any_defect"]["precision"], "{:.2f}")),
235        ("Recall at chosen confidence",
236         lambda r: cell(r["detection"]["any_defect"]["recall"], "{:.2f}")),
237        ("Defective packs flagged (sensitivity)",
238         lambda r: cell(r["verdicts"]["sensitivity"], "{:.2f}")),
239        ("Clean packs passed (specificity)",
240         lambda r: cell(r["verdicts"]["specificity"], "{:.2f}")),
241        ("Balanced accuracy of verdicts",
242         lambda r: cell(r["verdicts"]["balanced_accuracy"], "{:.2f}")),
243        ("Learned parameters", lambda r: cell(r["parameters"], "{:,}")),
244        ("Training time (minutes)",
245         lambda r: cell(r["training_seconds"] and r["training_seconds"] / 60, "{:.1f}")),
246        ("Time per photograph (milliseconds)",
247         lambda r: cell(r["seconds_per_photograph"] * 1000, "{:.1f}")),
248    ]
249    lines = ["| Measure | " + " | ".join(results) + " |", "| --- |" + " ---: |" * len(results)]
250    for name, value in rows:
251        lines.append(f"| {name} | " + " | ".join(value(r) for r in results.values()) + " |")
252    table = "\n".join(lines)
253    (arguments.artefacts / "comparison.md").write_text(table + "\n", encoding="utf-8")
254    plots.plot_precision_recall(
255        {name: result["detection"]["any_defect"] for name, result in results.items()},
256        arguments.figures / "precision-recall-comparison.svg",
257    )
258    print(table)

270def command_profile(arguments: argparse.Namespace) -> None:
271    """Time the detector and find where the time goes."""
272    class_names = read_class_names(arguments.dataset)
273    data = load_dataset(arguments.dataset)
274    lines = []
275
276    # Two passes are enough to see where training time goes.
277    detector = GridDetector(class_names, Settings(epochs=2, seed=arguments.seed))
278    profiler = cProfile.Profile()
279    profiler.enable()
280    detector.fit(data["train"], arguments.cache / "train.npz")
281    profiler.disable()
282    lines.append("Training (two passes): the ten functions with most time inside them\n")
283    lines.append(_profile_table(profiler))
284
285    photographs = data["test"].image_paths
286    profiler = cProfile.Profile()
287    start = time.perf_counter()
288    profiler.enable()
289    detector.detect(photographs)
290    profiler.disable()
291    seconds = (time.perf_counter() - start) / len(photographs)
292    lines.append(f"\nDetection on {len(photographs)} photographs read from disk\n")
293    lines.append(_profile_table(profiler))
294
295    network = detector.network
296    cells = detector.grid.rows * detector.grid.columns
297    lines.append(
298        f"\nTime per photograph: {seconds * 1000:.1f} milliseconds\n"
299        f"Network layers: {network.layer_sizes}\n"
300        f"Learned parameters: {network.number_of_parameters():,}\n"
301        f"Floating point operations per grid cell: {network.operations_per_prediction():,}\n"
302        f"Floating point operations per photograph ({cells} cells): "
303        f"{network.operations_per_prediction() * cells:,}\n"
304    )
305    report = "".join(lines)
306    (arguments.artefacts / "profile.txt").write_text(report, encoding="utf-8")
307    print(report)

310def _profile_table(profiler: cProfile.Profile) -> str:
311    """Format the ten most expensive functions recorded by a profiler."""
312    stream = io.StringIO()
313    pstats.Stats(profiler, stream=stream).strip_dirs().sort_stats("tottime").print_stats(10)
314    return stream.getvalue()

320def build_parser() -> argparse.ArgumentParser:
321    """Describe the commands and their options."""
322    parser = argparse.ArgumentParser(
323        prog="packaging-defects",
324        description="Detect pork rasher packaging defects in photographs.",
325    )
326    parser.add_argument("--dataset", type=Path, default=Path("dataset"),
327                        help="folder holding data.yaml and the train, valid and test folders")
328    parser.add_argument("--artefacts", type=Path, default=Path("artefacts"),
329                        help="folder for the trained detector and result files")
330    parser.add_argument("--figures", type=Path, default=Path("figures"),
331                        help="folder for figures")
332    parser.add_argument("--cache", type=Path, default=Path("cache"),
333                        help="folder for intermediate files that can be rebuilt")
334    parser.add_argument("--seed", type=int, default=0, help="seed for everything random")
335    commands = parser.add_subparsers(dest="command", required=True)
336
337    commands.add_parser("train", help=command_train.__doc__).set_defaults(run=command_train)
338    commands.add_parser("evaluate", help=command_evaluate.__doc__).set_defaults(
339        run=command_evaluate
340    )
341    baseline = commands.add_parser("baseline", help=command_baseline.__doc__)
342    baseline.add_argument("--epochs", type=int, default=30, help="passes through the data")
343    baseline.add_argument("--device", default=None,
344                          help="'mps' on Apple silicon, 'cpu', or a graphics card number")
345    baseline.add_argument("--weights", default=None,
346                          help="score an already trained model instead of training")
347    baseline.set_defaults(run=command_baseline)
348    commands.add_parser("compare", help=command_compare.__doc__).set_defaults(run=command_compare)
349    study = commands.add_parser("study", help=command_study.__doc__)
350    study.add_argument("--workers", type=int, default=2, help="trainings to run at once")
351    study.set_defaults(run=command_study)
352    commands.add_parser("profile", help=command_profile.__doc__).set_defaults(run=command_profile)
353    return parser
